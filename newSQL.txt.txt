CREATE DATABASE IF NOT EXISTS EJAR;
USE EJAR;

-- =========================================
-- DROP TABLES (ORDER MATTERS)
-- =========================================
DROP TABLE IF EXISTS MAKES;
DROP TABLE IF EXISTS PAYMENT;
DROP TABLE IF EXISTS RENTAL;
DROP TABLE IF EXISTS EMPLOYEE;
DROP TABLE IF EXISTS EQUIPMENT;
DROP TABLE IF EXISTS CUSTOMER;

-- =========================================
-- TABLE DEFINITIONS (MATCHING THE DIAGRAM)
-- =========================================

-- CUSTOMER
CREATE TABLE CUSTOMER (
    Cus_name   VARCHAR(80) PRIMARY KEY,
    Cus_mobile VARCHAR(20) NOT NULL,
    Email      VARCHAR(80) NOT NULL UNIQUE
);

-- EQUIPMENT
CREATE TABLE EQUIPMENT (
    Equip_id INT PRIMARY KEY,
    Type     VARCHAR(50) NOT NULL,
    Model    VARCHAR(80) NOT NULL,
    Price    DECIMAL(10,2) NOT NULL
);

-- EMPLOYEE
-- 
CREATE TABLE EMPLOYEE (
    Emp_id   INT PRIMARY KEY,
    Emp_name VARCHAR(60) NOT NULL,
    Eq_id    INT,
    FOREIGN KEY (Eq_id) REFERENCES EQUIPMENT(Equip_id)
        ON UPDATE CASCADE
        ON DELETE SET NULL
);

-- RENTAL

CREATE TABLE RENTAL (
    Rental_id   INT PRIMARY KEY,
    Start_date  DATE NOT NULL,
    End_date    DATE NOT NULL,
    City        VARCHAR(50) NOT NULL,
    Zip_code    VARCHAR(10) NOT NULL,
    Employee_id INT,
    FOREIGN KEY (Employee_id) REFERENCES EMPLOYEE(Emp_id)
        ON UPDATE CASCADE
        ON DELETE SET NULL
);

-- PAYMENT
CREATE TABLE PAYMENT (
    Invoice      INT PRIMARY KEY,
    Pay_date     DATE NOT NULL,
    Total_price  DECIMAL(10,2) NOT NULL,
    P_rental     INT NOT NULL,
    FOREIGN KEY (P_rental) REFERENCES RENTAL(Rental_id)
        ON UPDATE CASCADE
        ON DELETE CASCADE
);

-- MAKES (Customer rents specific equipment through a rental)
CREATE TABLE MAKES (
    C_name  VARCHAR(80) NOT NULL,
    Equ_id  INT NOT NULL,
    Rent_id INT NOT NULL,
    PRIMARY KEY (C_name, Equ_id, Rent_id),

    FOREIGN KEY (C_name) REFERENCES CUSTOMER(Cus_name)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    FOREIGN KEY (Equ_id) REFERENCES EQUIPMENT(Equip_id)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    FOREIGN KEY (Rent_id) REFERENCES RENTAL(Rental_id)
        ON UPDATE CASCADE
        ON DELETE RESTRICT
);

-- CREATE VIEW

CREATE VIEW equipment1 AS
SELECT 
    e.Equip_id,
    e.Type,
    e.Model,
    e.Price,

    CASE 
        WHEN m.Equ_id IS NULL THEN 'available'
        ELSE 'rented'
    END AS Status

FROM EQUIPMENT e
LEFT JOIN MAKES m 
    ON e.Equip_id = m.Equ_id;


CREATE VIEW rental1 AS
SELECT 
    r.Rental_id AS Rental_id,
    r.Start_date,
    r.End_date,
    r.City,
    r.Zip_code,
    p.Total_price AS Tot_price,
    m.Equ_id AS Equip_id,
    m.C_name ,
    r.Employee_id
FROM rental r
JOIN makes m ON r.Rental_id = m.Rent_id
JOIN payment p ON p.P_rental = m.Rent_id;
